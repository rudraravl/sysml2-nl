% ------------------------------------------------------
\begin{frame}{Objective}
\begin{itemize}
  \item Convert large SysML v1 models (\texttt{.mdzip}, \texttt{.xml}) to high-quality SysML v2 \emph{textual} artifacts.
  \item Produce \textbf{many small, semantically coherent files} (target: 200--500 lines each).
  \item Generate paired \textbf{NL summaries} for each file for a SysMLv2--NL dataset.
  \item Keep the process \textbf{deterministic} and \textbf{auditable}; use LLMs only for stylistic polish.
\end{itemize}
\end{frame}

% ------------------------------------------------------
\begin{frame}{Pipeline Overview}
\begin{enumerate}
  \item \textbf{Ingest v1}: unzip \texttt{.mdzip}/parse XML; normalize profiles/stereotypes.
  \item \textbf{Extract IR}: JSON intermediate representation (structure, behavior, requirements).
  \item \textbf{Rule-based mapping}: deterministic v1$\rightarrow$v2 text.
  \item \textbf{Sharding}: split into 200–500 line cohesive files.
  \item \textbf{LLM pass}: refine style, not semantics.
  \item \textbf{Validation}: syntax + semantic checks, imports/exports.
  \item \textbf{Packaging}: NL summaries + manifest (.jsonl).
\end{enumerate}
\end{frame}

% ------------------------------------------------------
\begin{frame}{v1$\rightarrow$v2 Mapping Rules}
\begin{columns}[T]
\begin{column}{0.5\linewidth}
\textbf{Structure}
\begin{itemize}
  \item Block $\rightarrow$ \texttt{part def}
  \item Part property $\rightarrow$ \texttt{part} membership
  \item Value property $\rightarrow$ \texttt{attribute}/constraint
  \item Connector $\rightarrow$ \texttt{connection}
\end{itemize}
\textbf{Requirements}
\begin{itemize}
  \item Requirement $\rightarrow$ \texttt{requirement} def
  \item Links $\rightarrow$ v2 allocations/satisfy relations
\end{itemize}
\end{column}
\begin{column}{0.5\linewidth}
\textbf{Behavior}
\begin{itemize}
  \item Activity $\rightarrow$ \texttt{action def}s
  \item Sequence $\rightarrow$ occurrences/messages
  \item Use case $\rightarrow$ \texttt{usecase def}s
\end{itemize}
\textbf{Stereotypes}
\begin{itemize}
  \item $\rightarrow$ v2 \texttt{metadata} / \texttt{\#keywords}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% ------------------------------------------------------
\begin{frame}{Sharding Strategy (200--500 lines each)}
\begin{itemize}
  \item Split by concern: Structure / Behavior / Requirements / Interfaces.
  \item Within each concern, anchor on top block or scenario root.
  \item Greedy BFS add elements until $\sim$400 lines.
  \item Record \texttt{exports}/\texttt{imports}.
  \item Keep large units/enums in shared \texttt{Libraries/}.
\end{itemize}
\end{frame}

\begin{frame}{LLM Role (Polish Only)}
\begin{itemize}
  \item Input: deterministic v2 + IR snippet
  \item Output: stylistically refined v2 text or short NL summary
  \item Guardrail prompt:
  \begin{quote}
  ``Do not invent new elements. Only reorganize or reformat existing definitions for v2 idiom.''
  \end{quote}
\end{itemize}
\end{frame}
