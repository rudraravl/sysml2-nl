This model describes a specialized quadcopter drone designed for forest fire observation and surveillance.

The drone is built with a standard quad-rotor airframe, consisting of a central body, four arms, and landing gear. It is propelled by four engines, with two propellers rotating clockwise and two counter-clockwise to provide lift and stable, maneuverable flight.

Engineered for extended missions, the drone is equipped with a high-capacity, long-distance battery. A central power management module is responsible for monitoring the battery's charge status and distributing electricity to the four engines and other electronic components.

The drone's "brain" is a flight control unit that manages all aspects of its operation. It relies on a suite of sensors for navigation and stability: a GPS module for determining its location, an Inertial Measurement Unit (IMU) for sensing orientation and movement, and a barometer for precise altitude control. Crucially, the flight control unit constantly receives battery status updates from the power management module, allowing it to manage flight time and execute return-to-base procedures when power is low.

For its primary observation role, the drone is fitted with a camera. It also includes an obstacle avoidance sensor to enhance safety during flight. A complete communication system, with a radio transmitter, receiver, and antenna, enables it to be controlled remotely and transmit data, such as a live video feed, back to an operator.

		private import ScalarValues::*;

		abstract value type ScalarValue;
		value type Voltage :> ScalarValue specializes Real {
			attribute value: Real;
		}

			attribute value: Real;
		}
		value type Altitude :> ScalarValue specializes Real {
			attribute value: Real;
		}
		value type RotationalSpeed :> ScalarValue specializes Real {
			attribute value: Real;
		}
		value type Throttle :> ScalarValue specializes Real {
			attribute value: Real;
		}

		item def ElectricalPower {
			attribute voltage : Voltage;
		}
		item def AltitudeData {
			attribute reading : Altitude;
		}
		item def MotorControlSignal {
			attribute throttleValue : Throttle;
		}
	}

	package DroneSystem {
		import DroneDomain::*;

		part def Battery {
			attribute capacity : Capacity = 5000.0;
			attribute currentVoltage : Voltage = 14.8;
			port powerSource {
				out item power : ElectricalPower;
			}
			flow of power from powerSource.power {
				voltage = currentVoltage
			}
		}

		part def Motor {
			attribute currentSpeed : RotationalSpeed;
			port powerInput {
				in item power : ElectricalPower;
			}
			port controlInput {
				in item signal : MotorControlSignal;
			}
			constraint speedControl {
				currentSpeed.value == signal.throttleValue.value * 250.0
			}
		}

		part def AltitudeSensor {
			attribute measuredAltitude : Altitude;
			port powerInput {
				in item power : ElectricalPower;
			}
			port dataOutput {
				out item data : AltitudeData;
			}
			flow of data from dataOutput.data {
				reading = measuredAltitude
			}
		}

		part def PowerDistribution {
			port batteryInput {
				in item power : ElectricalPower;
			}
			port controllerOutput {
				out item power : ElectricalPower;
			}
			port motorOutputs[4] {
				out item power : ElectricalPower;
			}
			flow of power from batteryInput.power to controllerOutput.power;
			for i in 1..4 {
				flow of power from batteryInput.power to motorOutputs[i].power;
			}
		}

		part def FlightController {
			attribute targetAltitude : Altitude = 20.0;
			port powerInput {
				in item power : ElectricalPower;
			}
			port sensorInput {
				in item data : AltitudeData;
			}
			port motorControl[4] {
				out item signal : MotorControlSignal;
			}

			action def maintainAltitude {
				in currentAltitude : Altitude;
				in target : Altitude;
				out throttle[4] : Throttle;

				let altitudeError = target.value - currentAltitude.value;
				let baseThrottle = 55.0;
				let adjustment = altitudeError * 2.5;
				for i in 1..4 {
					throttle[i].value = baseThrottle + adjustment;
				}
			}

			state machine FlightBehavior {
				initial state Grounded;
				state Flying {
					state Hovering {
						perform action maintainAltitude(
							currentAltitude = sensorInput.data.reading,
							target = targetAltitude,
							throttle = motorControl.signal.throttleValue
						);
					}
				}
				transition from Grounded to Flying on takeoff;
				transition from Flying to Grounded on land;
			}
		}

		part def Drone {
			part battery : Battery;
			part motor : Motor[4];
			part altitudeSensor : AltitudeSensor;
			part flightController : FlightController;
			part powerDistribution : PowerDistribution;

			connect battery.powerSource to powerDistribution.batteryInput;
			connect powerDistribution.controllerOutput to flightController.powerInput;
			connect powerDistribution.controllerOutput to altitudeSensor.powerInput;
			for i in 1..4 {
				connect powerDistribution.motorOutputs[i] to motor[i].powerInput;
			}

			connect altitudeSensor.dataOutput to flightController.sensorInput;
			for i in 1..4 {
				connect flightController.motorControl[i] to motor[i].controlInput;
			}
		}
	}

		}

	}
