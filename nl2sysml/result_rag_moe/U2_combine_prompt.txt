package DroneSystem {
	import ISQ::Units::*;
	import ScalarValues::*;

	// Value and Item Type Definitions
	value def Voltage redefines RealValue {
		in unit :> V;
	}

	value def Capacity redefines RealValue {
		in unit :> A * h;
	}

	value def Altitude redefines RealValue {
		in unit :> m;
	}

	value def RotationalSpeed redefines RealValue {
		// Using Hertz (1/s) for RPM (revolutions per minute) is unconventional,
		// but standard units are preferred. Here we assume a conversion factor is applied.
		in unit :> Hz;
	}

	item def PowerSignal {
		attribute voltage : Voltage;
	}

	item def AltitudeData {
		attribute reading : Altitude;
	}

	item def MotorCommand {
		attribute targetSpeed : RotationalSpeed;
	}

	// Part Definitions
	part def Battery {
		attribute maxCapacity : Capacity = 5[A*h];
		attribute currentCharge : Capacity;
		attribute outputVoltage : Voltage = 14.8[V];
		out item port powerOut : PowerSignal;
		bind powerOut.voltage to outputVoltage;
	}

	part def PowerDistributionUnit {
		in item port powerIn : PowerSignal;
		out item port motorPowerOut[4] : PowerSignal;
		out item port fcPowerOut : PowerSignal;
		out item port sensorPowerOut : PowerSignal;
		bind motorPowerOut.voltage to powerIn.voltage;
		bind fcPowerOut.voltage to powerIn.voltage;
		bind sensorPowerOut.voltage to powerIn.voltage;
	}

	part def Motor {
		in item port powerIn : PowerSignal;
		in item port commandIn : MotorCommand;
		attribute currentSpeed : RotationalSpeed;

		action updateSpeed {
			in command : MotorCommand;
			inout currentSpeed;
			accept command then {
				currentSpeed = command.targetSpeed;
			}
		}
	}

	part def AltitudeSensor {
		in item port powerIn : PowerSignal;
		out item port dataOut : AltitudeData;
		attribute measuredAltitude : Altitude;

		perform action measure {
			out data : AltitudeData;
			first start;
			then action senseAltitude;
			then action outputData {
				data.reading = measuredAltitude;
			}
			then done;
		}
	}

	part def FlightController {
		in item port powerIn : PowerSignal;
		in item port altitudeSenseIn : AltitudeData;
		out item port motorCommandOut[4] : MotorCommand;
		attribute targetAltitude : Altitude = 20[m];
		attribute isStable : Boolean;

		action calculateMotorAdjustment {
			in currentAlt : Altitude;
			in targetAlt : Altitude;
			out speedAdjustment : RotationalSpeed;
		}

		state machine FlightSM {
			initial state Grounded;
			state Climbing;
			state Hovering;
			state Landing;

			transition from Grounded to Climbing on takeOffCmd;

			transition from Climbing to Hovering
				when altitudeSenseIn.reading.value >= targetAltitude.value
				do {
					isStable = true;
				};

			transition from Hovering to Climbing
				when altitudeSenseIn.reading.value < targetAltitude.value - 0.5
				do {
					isStable = false;
				};

			transition from Hovering to Landing on landCmd;

			transition from Landing to Grounded
				when altitudeSenseIn.reading.value <= 0.1;
		}
	}

	// Top-level Drone Assembly
	part def Drone {
		part battery : Battery;
		part pdu : PowerDistributionUnit;
		part motors[4] : Motor;
		part altimeter : AltitudeSensor;
		part flightController : FlightController;

		connect battery.powerOut to pdu.powerIn;
		connect pdu.motorPowerOut to motors.powerIn;
		connect pdu.fcPowerOut to flightController.powerIn;
		connect pdu.sensorPowerOut to altimeter.powerIn;
		connect altimeter.dataOut to flightController.altitudeSenseIn;
		connect flightController.motorCommandOut to motors.commandIn;
	}

	// Requirements
	requirement <'DR-REQ-01'> stableHover {
		doc "The drone shall be capable of maintaining a stable hover at a commanded altitude within +/- 0.5 meters in calm wind conditions.";
		subject drone : Drone;
		assert constraint {
			drone.flightController.isStable implies
			abs(drone.altimeter.measuredAltitude.value - drone.flightController.targetAltitude.value) <= 0.5
		}
	}

	requirement <'SYS-REQ-10'> sensorAccuracy {
		doc "The altitude sensor shall provide measurements with an accuracy of +/- 0.2 meters.";
		subject altimeter : AltitudeSensor;
	}

	// Derivation of Requirements
	#derivation connection {
		end #original ::> stableHover;
		end #derive ::> sensorAccuracy;
	}

	// System Instantiation
	part simpleDrone : Drone;

package DroneSystem {
    import ScalarValues::*;
    import ISQ::*;
    import SI::*;
    
    attribute def VoltageValue :> Real {
        attribute :>> num : Real;
        attribute unit : VoltageUnit;
    }
    
    attribute def CurrentValue :> Real {
        attribute :>> num : Real;
        attribute unit : CurrentUnit;
    }
    
    attribute def CapacityValue :> Real {
        attribute :>> num : Real;
        attribute unit : EnergyUnit;
    }
    
    attribute def AltitudeValue :> Real {
        attribute :>> num : Real;
        attribute unit : LengthUnit;
    }
    
    attribute def RPMValue :> Real {
        attribute :>> num : Real;
    }
    
    attribute def PowerValue :> Real {
        attribute :>> num : Real;
        attribute unit : PowerUnit;
    }
    
    enum def BatteryStateKind {
        enum full;
        enum normal;
        enum low;
        enum critical;
        enum depleted;
    }
    
    enum def MotorStateKind {
        enum off;
        enum starting;
        enum running;
        enum stopping;
        enum fault;
    }
    
    item def ElectricalPower {
        attribute voltage : VoltageValue;
        attribute current : CurrentValue;
    }
    
    item def SensorData {
        attribute altitude : AltitudeValue;
        attribute timestamp : Real;
    }
    
    item def ControlSignal {
        attribute targetRPM : RPMValue;
        attribute enabled : Boolean;
    }
    
    part def Battery {
        attribute capacity : CapacityValue;
        attribute currentCharge : CapacityValue;
        attribute voltage : VoltageValue;
        attribute maxCurrent : CurrentValue;
        attribute stateOfCharge : Real;
        
        port powerOut {
            out item electricalPower : ElectricalPower;
        }
        
        port statusPort {
            out attribute batteryState : BatteryStateKind;
            out attribute chargeLevel : Real;
        }
        
        state batteryStateMachine {
            entry; then full;
            
            state full {
                entry assign stateOfCharge = 1.0;
                entry assign batteryState = BatteryStateKind::full;
            }
            
            state normal {
                entry assign batteryState = BatteryStateKind::normal;
            }
            
            state low {
                entry assign batteryState = BatteryStateKind::low;
            }
            
            state critical {
                entry assign batteryState = BatteryStateKind::critical;
            }
            
            state depleted {
                entry assign batteryState = BatteryStateKind::depleted;
                entry assign stateOfCharge = 0.0;
            }
            
            transition full_to_normal
                first full
                accept at stateOfCharge < 0.9
                then normal;
                
            transition normal_to_low
                first normal
                accept at stateOfCharge < 0.3
                then low;
                
            transition low_to_critical
                first low
                accept at stateOfCharge < 0.15
                then critical;
                
            transition critical_to_depleted
                first critical
                accept at stateOfCharge < 0.05
                then depleted;
        }
        
        action dischargeBattery {
            in powerDemand : PowerValue;
            out updatedCharge : CapacityValue;
            
            first start;
            then action calculateDischarge {
                in demand :>> powerDemand;
                out newCharge : CapacityValue;
            }
            then action updateState {
                in charge :>> calculateDischarge.newCharge;
            }
            then done;
        }
    }
    
    part def Motor {
        attribute maxRPM : RPMValue;
        attribute currentRPM : RPMValue;
        attribute powerConsumption : PowerValue;
        attribute efficiency : Real;
        
        port powerIn {
            in item electricalPower : ElectricalPower;
        }
        
        port controlPort {
            in item controlSignal : ControlSignal;
        }
        
        port statusPort {
            out attribute motorState : MotorStateKind;
            out attribute rpm : RPMValue;
        }
        
        state motorStateMachine {
            entry; then off;
            
            state off {
                entry assign motorState = MotorStateKind::off;
                entry assign currentRPM.num = 0.0;
            }
            
            state starting {
                entry assign motorState = MotorStateKind::starting;
                do action rampUp {
                    first start;
                    then action increaseRPM;
                    then done;
                }
            }
            
            state running {
                entry assign motorState = MotorStateKind::running;
                do action maintainSpeed {
                    first start;
                    then action regulateRPM;
                    then done;
                }
            }
            
            state stopping {
                entry assign motorState = MotorStateKind::stopping;
                do action rampDown {
                    first start;
                    then action decreaseRPM;
                    then done;
                }
            }
            
            state fault {
                entry assign motorState = MotorStateKind::fault;
            }
            
            transition off_to_starting
                first off
                accept controlSignal.enabled == true
                then starting;
                
            transition starting_to_running
                first starting
                accept at currentRPM.num >= controlSignal.targetRPM.num * 0.95
                then running;
                
            transition running_to_stopping
                first running
                accept controlSignal.enabled == false
                then stopping;
                
            transition stopping_to_off
                first stopping
                accept at currentRPM.num <= 0.0
                then off;
                
            transition any_to_fault
                first off, starting, running, stopping
                accept electricalPower.voltage.num < 10.0
                then fault;
        }
    }
    
    part def AltitudeSensor {
        attribute measurementRange : AltitudeValue;
        attribute accuracy : AltitudeValue;
        attribute currentAltitude : AltitudeValue;
        attribute samplingRate : Real;
        
        port powerIn {
            in item electricalPower : ElectricalPower;
        }
        
        port dataOut {
            out item sensorData : SensorData;
        }
        
        action measureAltitude {
            out measurement : SensorData;
            
            first start;
            then action readSensor {
                out rawValue : AltitudeValue;
            }
            then action calibrateReading {
                in raw :>> readSensor.rawValue;
                out calibrated : AltitudeValue;
            }
            then action packageData {
                in altitude :>> calibrateReading.calibrated;
                out data :>> measurement;
            }
            then done;
        }
        
        state sensorStateMachine {
            entry; then idle;
            
            state idle;
            
            state measuring {
                do measureAltitude;
            }
            
            state transmitting {
                do action sendData {
                    first start;
                    then action transmit;
                    then done;
                }
            }
            
            transition idle_to_measuring
                first idle
                then measuring;
                
            transition measuring_to_transmitting
                first measuring
                then transmitting;
                
            transition transmitting_to_idle
                first transmitting
                then idle;
        }
    }
    
    part def DroneController {
        port batteryStatusIn {
            in attribute batteryState : BatteryStateKind;
            in attribute chargeLevel : Real;
        }
        
        port motorStatusIn {
            in attribute motorState : MotorStateKind;
            in attribute rpm : RPMValue;
        }
        
        port altitudeDataIn {
            in item sensorData : SensorData;
        }
        
        port motorControlOut {
            out item controlSignal : ControlSignal;
        }
        
        action controlLoop {
            first start;
            then action readSensors {
                out altitude : AltitudeValue;
                out batteryLevel : Real;
            }
            then action computeControl {
                in currentAltitude :>> readSensors.altitude;
                in battery :>> readSensors.batteryLevel;
                out motorCommand : ControlSignal;
            }
            then action sendCommands {
                in command :>> computeControl.motorCommand;
            }
            loop back computeControl;
        }
    }
    
    part drone {
        part battery : Battery {
            :>> capacity.num = 5000.0;
            :>> currentCharge.num = 5000.0;
            :>> voltage.num = 14.8;
            :>> maxCurrent.num = 30.0;
            :>> stateOfCharge = 1.0;
        }
        
        part motor : Motor {
            :>> maxRPM.num = 8000.0;
            :>> currentRPM.num = 0.0;
            :>> powerConsumption.num = 150.0;
            :>> efficiency = 0.85;
        }
        
        part altitudeSensor : AltitudeSensor {
            :>> measurementRange.num = 500.0;
            :>> accuracy.num = 0.5;
            :>> currentAltitude.num = 0.0;
            :>> samplingRate = 10.0;
        }
        
        part controller : DroneController;
        
        connect battery.powerOut to motor.powerIn;
        connect battery.powerOut to altitudeSensor.powerIn;
        connect battery.statusPort to controller.batteryStatusIn;
        connect motor.statusPort to controller.motorStatusIn;
        connect altitudeSensor.dataOut to controller.altitudeDataIn;
        connect controller.motorControlOut to motor.controlPort;
    }
    
    requirement <'REQ-001'> droneOperationalRequirement {
        doc /* The drone shall maintain stable operation with 
             * integrated battery, motor, and altitude sensing.
             */
        subject drone : drone;
        
        require batteryCapacityReq;
        require motorPerformanceReq;
        require altitudeSensingReq;
    }
    
    requirement <'REQ-002'> batteryCapacityReq {
        doc /* The battery shall provide sufficient capacity 
             * for extended flight operations.
             */

        assert constraint {
            battery.capacity.num >= 4000.0
        }
    }
    
    requirement <'REQ-003'> motorPerformanceReq {
        doc /* The motor shall achieve operational RPM range 
             * for flight control.
             */
        subject motor :>> drone.motor;
        
        assert constraint {
            motor.maxRPM.num >= 6000.0
        }
    }
    
    requirement <'REQ-004'> altitudeSensingReq {
        doc /* The altitude sensor shall provide accurate 
             * altitude measurements within operational range.
             */
        subject sensor :>> drone.altitudeSensor;
        
        assert constraint {
            sensor.accuracy.num <= 1.0 and
            sensor.measurementRange.num >= 300.0
        }
    }
    
    satisfy droneOperationalRequirement by drone;
}
