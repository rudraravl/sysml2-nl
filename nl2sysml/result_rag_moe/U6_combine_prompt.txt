The system's primary function is to automatically charge a battery to its full capacity. It operates using a continuous feedback loop. package 'Water Pump System' {

	private import SI_Units::*;

	value def Volume :> Real { unit = L; }

	item def Water;

	port def WaterInPort {
		in item water : Water;
	}

	port def WaterOutPort {
		out item water : Water;
	}

	part def WaterTank {
		attribute level : Percentage = 10.0;
		attribute capacity : Volume = 1000.0;
		port waterIn : WaterInPort;
	}

	part def WaterPump {
		port waterOut : WaterOutPort;
		in reception start;
		in reception stop;

		state machine PumpState {
			initial state Off;
			state On;

			transition from Off to On on start;
			transition from On to Off on stop;
		}
	}

	requirement def ActivationRequirement {
		doc "The pump shall activate when the tank level is below a specified threshold.";
	}

	requirement def DeactivationRequirement {
		doc "The pump shall deactivate when the tank level is at or above a specified threshold.";
	}

	part def WaterPumpSystem {
		part tank : WaterTank;
		part pump : WaterPump;

		connect pump.waterOut to tank.waterIn;

		state machine ControlLogic {
			initial state Idle;
			state Pumping;

			transition from Idle to Pumping
				when tank.level < 30.0
				do pump.start;

			transition from Pumping to Idle
				when tank.level >= 90.0
				do pump.stop;
		}

		satisfy activationReq by ControlLogic;
		satisfy deactivationReq by ControlLogic;
	}

	requirement activationReq : ActivationRequirement {
		doc "The pump shall activate when the tank level is below 30%.";
		subject system.pump;
	}

	requirement deactivationReq : DeactivationRequirement {
		doc "The pump shall deactivate when the tank level is at or above 90%.";
		subject system.pump;
	}

	part system : WaterPumpSystem;

		attribute level : WaterLevel;
		in item waterSupply : Water;

		attribute currentLevel : WaterLevel;
		port tankPort : WaterOutPort {
			attribute redefines level = currentLevel;
		}
		in item waterIn : Water;
		out item waterOut : Water;
	}
	
	part def Pump {
		attribute isActive : Boolean;
		port pumpInPort : WaterInPort;
		out item pumpOut : Water;
	}
	
	part def LevelSensor {
		attribute measuredLevel : WaterLevel;
		out signal levelReading : WaterLevel;
	}
	
	part def Controller {
		in signal tankLevel : WaterLevel;
		out signal pumpCommand : Boolean;
	}
	
	action def MonitorTankLevel {
		in tankLevel : WaterLevel;
		out currentLevel : WaterLevel;
	}
	
	action def ActivatePump {
		out pumpState : Boolean = true;
	}
	
	action def DeactivatePump {
		out pumpState : Boolean = false;
	}
	
	action def ControlPumpOperation {
		in levelInput : WaterLevel;
		out pumpControl : Boolean;
		
		first start;
		then merge checkLevel;
		
		then action monitor : MonitorTankLevel {
			in tankLevel = levelInput;
			out currentLevel : WaterLevel;
		}
		
		then decide;
			if monitor.currentLevel < 30.0 then activatePumpAction;
			if monitor.currentLevel >= 90.0 then deactivatePumpAction;
			if monitor.currentLevel >= 30.0 and monitor.currentLevel < 90.0 then maintainState;
		
		action activatePumpAction : ActivatePump {
			out pumpState;
		}
		then assign pumpControl = activatePumpAction.pumpState;
		then checkLevel;
		
		action deactivatePumpAction : DeactivatePump {
			out pumpState;
		}
		then assign pumpControl = deactivatePumpAction.pumpState;
		then checkLevel;
		
		action maintainState {
			out holdState : Boolean = pumpControl;
		}
		then checkLevel;
	}
	
	state def PumpControlStateMachine {
		in attribute tankLevel : WaterLevel;
		out attribute pumpActive : Boolean;
		
		entry; then idle;
		
		state idle {
			entry assign pumpActive = false;
		}
		
		transition idleToActivating
			first idle
			accept at when tankLevel < 30.0
			then activating;
		
		state activating {
			entry assign pumpActive = true;
		}

			}

		}

			first deactivating
