		}

			in flow reading : WindSpeed;
		}

		}

			in p_humidity : RelativeHumidity;
			in p_wind : WindSpeed;
			out r_report : WeatherReport;
		}
	}

		port reportIn {
			in flow report : WeatherReport;
		}
		action def transmitReport {
			in r_report : WeatherReport;
		}
	}

			}
