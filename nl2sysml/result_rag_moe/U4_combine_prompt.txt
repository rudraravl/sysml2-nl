package 'RoboticArmSystem' {
	private import ISQ::*;
	private import SI::*;

	/*
	 * Value and Item type definitions used throughout the system model.
	 */
	value def AngleValue :> PlaneAngleValue;

	item def SmallObject {
		attribute mass : MassValue;
	}

	/*
	 * Interface definitions for control and physical connections.
	 */
	interface def ArmControlInterface {
		inout action performPickAndPlace(
			in pickupAngle : AngleValue,
			in placeAngle : AngleValue
		);
	}

	interface def MechanicalInterface;

	/*
	 * Requirement definitions for the robotic arm system.
	 */
	requirement def RotationRequirement {
		id = "RA-REQ-01";
		text = "The robot arm base shall be capable of 360-degree rotation.";
	}

	requirement def GraspingRequirement {
		id = "RA-REQ-02";
		text = "The gripper must be able to securely grasp and release a SmallObject.";
	}

	/*
	 * Part definitions for the individual components of the robotic arm.
	 */
	part def RotationalBase {
		port mountPoint : MechanicalInterface;

		action rotate(in targetAngle : AngleValue);

		attribute currentAngle : AngleValue {
			inout;
			default = 0['°'];
		}

		assert constraint AngleLimit {
			doc "The physical rotation is limited to a single 360-degree circle.";
			currentAngle >= 0['°'] and currentAngle <= 360['°']
		}
	}

	part def ArmLinkage {
		port baseMount : ~MechanicalInterface;
		port gripperMount : MechanicalInterface;
	}

	part def Gripper {
		port mountPoint : ~MechanicalInterface;
		port objectHandler {
			inout item payload : SmallObject[0..1];
		}

		action open;
		action close;

		state machine GripperState {
			initial state Open;
			state Closed;

			transition from Open to Closed on close {
				doc "When the close command is received, the gripper closes around an object.";
			}
			transition from Closed to Open on open {
				doc "When the open command is received, the gripper releases its payload.";
			}
		}
	}

	/*
	 * Top-level assembly of the RobotArm system, integrating all parts and behaviors.
	 */
	part def RobotArm {
		port control : ArmControlInterface;

		part base : RotationalBase;
		part linkage : ArmLinkage;
		part gripper : Gripper;

		action performPickAndPlace(in pickupAngle : AngleValue, in placeAngle : AngleValue) {
			doc "Orchestrates the sub-parts to perform a complete pick and place task.";
			
			// Sequence of calls to sub-part actions
			call gripper.open;
			call base.rotate(targetAngle = pickupAngle);
			call gripper.close;
			call base.rotate(targetAngle = placeAngle);
			call gripper.open;
		}

		connect base.mountPoint to linkage.baseMount;
		connect linkage.gripperMount to gripper.mountPoint;

		bind control.performPickAndPlace to self.performPickAndPlace;
	}

	/*
	 * Verification of requirements against the design.
	 */
	satisfy RotationRequirement by RobotArm::base {
		doc "The RotationalBase part, with its rotate action and defined angle attribute, is designed to meet the 360-degree rotation requirement.";
	}

	satisfy GraspingRequirement by RobotArm::gripper {
		doc "The Gripper part, with its open/close actions and state machine, directly satisfies the requirement to grasp and release objects.";
	}
}

		in angle : PlaneAngleValue;
