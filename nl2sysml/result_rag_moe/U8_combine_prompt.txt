	package Behaviors {
		private import Definitions::*;
		
		state machine def CoffeeMakerBehavior {
			port selectionPort;
			port statusPort;
			port heaterPort;
			port frotherPort;
			
			initial state Idle {
				entry do send new StatusUpdate(status := StatusType::Idle) via statusPort;
				transition on accept sel : UserSelection via selectionPort to HeatingWater;
			}
			
			state HeatingWater {
				entry do {
					send new StatusUpdate(status := StatusType::Heating) via statusPort;
					send new ControlSignal(command := Command::On) via heaterPort;
				}
				transition to Brewing after 25[s] {
					guard accept(sel).selection == SelectionType::Espresso;
				}
				transition to FrothingMilk after 25[s] {
					guard accept(sel).selection == SelectionType::Cappuccino or
					      accept(sel).selection == SelectionType::Latte;
				}
				exit do send new ControlSignal(command := Command::Off) via heaterPort;
			}
			
			state FrothingMilk {
				entry do {
					send new StatusUpdate(status := StatusType::Frothing) via statusPort;
					send new ControlSignal(command := Command::On) via frotherPort;
				}
				transition to Brewing after 15[s];
				exit do send new ControlSignal(command := Command::Off) via frotherPort;
			}
			
			state Brewing {
				entry do send new StatusUpdate(status := StatusType::Brewing) via statusPort;
				transition to Done after 20[s];
			}
			
			final state Done {
				entry do send new StatusUpdate(status := StatusType::Ready) via statusPort;
				transition to Idle after 60[s];
			}
		}
	}
