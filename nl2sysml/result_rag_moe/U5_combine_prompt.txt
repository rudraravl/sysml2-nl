    /*

            attribute latitude : ScalarValues::Real;
            attribute longitude : ScalarValues::Real;
            attribute altitude : SI::MetreValue;
        }

        value def ObstacleProximity {
            attribute distance : SI::MetreValue;
            attribute isDetected : ScalarValues::Boolean;
        }

        item def VideoStream;
        item def HighResolutionImage;

        // Component Definitions
        part def Airframe {
            attribute mass : SI::KilogramValue;
        }

        part def GpsModule {
            port location : GpsCoordinates;
            perform action acquireSignal {
                out location : GpsCoordinates;
            }
        }

        part def ObstacleSensor {
            port proximity : ObstacleProximity;
            perform action scanEnvironment {
                out proximity : ObstacleProximity;
            }
        }

        part def CameraModule {
            port videoFeed : VideoStream;
            port imageOut : HighResolutionImage;
            
            perform action captureImage {
                out result : HighResolutionImage;
            }
            
            perform action streamVideo {
                out stream : VideoStream;
            }
        }

        part def PropulsionUnit {
            attribute thrust : SI::NewtonValue;
            port powerIn;
            port commandIn;
        }

        part def PowerSystem {
            attribute capacity : SI::AmpereHourValue;
            attribute chargeLevel : ScalarValues::RatioValue;
            port powerDistribution[4];
        }

        part def FlightController {
            port gpsDataIn : GpsCoordinates;
            port obstacleDataIn : ObstacleProximity;
            port propulsionCommandOut[4];
            port cameraCommandOut;

            state operationalMode {
                entry; then Grounded;
                
                state Grounded;
                accept TakeoffSignal then Flying;
                
                state Flying {
                    entry / perform navigateToWaypoint;
                    state Navigating;
                    state Avoiding {
                        entry / perform executeAvoidanceManeuver;
                    }
                    transition Navigating_to_Avoiding 
                        when obstacleDataIn.isDetected and obstacleDataIn.distance < 5[SI::m]
                        first Navigating then Avoiding;
                    transition Avoiding_to_Navigating
                        when not obstacleDataIn.isDetected or obstacleDataIn.distance >= 5[SI::m]
                        first Avoiding then Navigating;
                }
                accept LandSignal then Grounded;
            }

            perform action navigateToWaypoint {
                in targetLocation : GpsCoordinates;
                in currentPosition : GpsCoordinates;
                out commands[4] :> propulsionCommandOut;
            }

            perform action executeAvoidanceManeuver {
                in obstacleData : ObstacleProximity;
                out commands[4] :> propulsionCommandOut;
            }
            
            satisfy ObstacleAvoidanceRequirement by self.executeAvoidanceManeuver;
        }
    }

     * Requirement definitions for the drone system.

        id = "REQ-NAV-001";
        text = "The drone system shall detect and actively avoid obstacles

    }

        part propulsionUnits[4] : DroneLibrary::PropulsionUnit;
        part gps : DroneLibrary::GpsModule;
        part camera : DroneLibrary::CameraModule;
        part obstacleSensor : DroneLibrary::ObstacleSensor;
        part controller : DroneLibrary::FlightController;
