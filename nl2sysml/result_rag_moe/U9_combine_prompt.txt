Example 1 NL:
This model represents the process of a digital camera taking a photograph. The system operates as a sequence of distinct actions that transform a real-world view into a final, viewable image.

The process is initiated by a user trigger, such as pressing the shutter button, which identifies the desired real-world **scene** to be captured. The first automated step is to **focus**, where the camera's lens system adjusts to create a sharp, clear optical **image** of that scene on its internal sensor.

Immediately following the focus, the camera performs the **shoot** action. This is the moment of capture, where the light and color data from the focused image is read by the sensor and processed into a digital **picture** file.

Finally, this newly created picture is sent to be **displayed**, allowing the user to instantly review the photo they have just taken, typically on an LCD screen on the back of the camera. Once displayed, the camera is ready to capture the next scene.

package 'Mobile App Login Process' {

	// Data and Value Definitions
	item def UserCredentials {
		attribute username : String;
		attribute password : String;
	}

	item def AuthenticationToken {
		attribute tokenValue : String;
	}

	item def LoginStatus {
		attribute success : Boolean;
		attribute message : String;
	}

	item def UserProfile {
		attribute name : String;
		attribute email : String;
		attribute lastLogin : String;
	}

	// Part Definitions for System and Actors
	part def User;

	part def MobileApp;

	part def AuthenticationServer;

	part def ProfileServer;

	part def LoginSystem {
		part app : MobileApp;
		part authServer : AuthenticationServer;
		part profileServer : ProfileServer;

		connect app to authServer;
		connect app to profileServer;
	}

	// Action Definitions for individual process steps
	action def EnterCredentials {
		out credentials : UserCredentials;
	}

	action def Authenticate {
		in credentials : UserCredentials;
		out token : AuthenticationToken;
		out status : LoginStatus;
	}

	action def RequestProfile {
		in token : AuthenticationToken;
		out profile : UserProfile;
	}

	action def DisplayProfile {
		in profile : UserProfile;
	}

	action def ShowLoginError {
		in status : LoginStatus;
	}

	// Main process definition orchestrating the actions
	action def RetrieveProfileProcess;

	action mainProcess : RetrieveProfileProcess {
		// Step 1: User provides credentials
		first action enterCreds : EnterCredentials {
			out credentials;
		}

		// Step 2: Credentials are sent for authentication
		flow from enterCreds.credentials to authenticateUser.credentials;
		then action authenticateUser : Authenticate {
			in credentials;
			out token;
			out status;
		}

		// Step 3: Decide next step based on authentication status
		then decision checkAuth(authenticateUser.status.success) {
			if true {
				// Step 4a: If successful, use token to request profile
				flow from authenticateUser.token to fetchProfile.token;
				then action fetchProfile : RequestProfile {
					in token;
					out profile;
				}

				// Step 5a: Display the retrieved user profile
				flow from fetchProfile.profile to display.profile;
				then action display : DisplayProfile {
					in profile;
				}

				// End of successful path
				then succeed;
			}
			else {
				// Step 4b: If failed, show an error message
				flow from authenticateUser.status to showError.status;
				then action showError : ShowLoginError {
					in status;
				}

				// End of failure path
				then fail;
			}
		}
	}

	// Requirements
	requirement def SecureAuthentication {
		doc "The system shall securely transmit and validate user credentials against a trusted authentication service.";
	}

	requirement def ProfileDataRetrieval {
		doc "A successfully authenticated user shall be able to retrieve their complete profile data.";
	}

	requirement authRequirement on Authenticate {
		satisfy SecureAuthentication;
	}

	requirement profileRequirement on RequestProfile {
		satisfy ProfileDataRetrieval;
	}
