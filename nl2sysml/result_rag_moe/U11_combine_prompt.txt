package 'Patient Monitoring System' {
	private import SI::*;
	private import ISQ::*;
	private import ScalarValues::*;
	private import KerML::Triggers::*;
	private import KerML::Time::*;

	// Custom Units and Value Types for Medical Domain
	unit beatsPerMinute;
	unit millimetersOfMercury;

	value def HeartRateValue :> Real {
		in unit beatsPerMinute;
	}

	value def BloodPressureValue :> Real {
		in unit millimetersOfMercury;
	}

	value def OxygenSaturationValue :> Percentage;
	value def BodyTemperatureValue :> TemperatureValue;

	// Data structure for communicating vital signs
	item def VitalSignsData {
		attribute heartRate : HeartRateValue;
		attribute bloodPressureSystolic : BloodPressureValue;
		attribute bloodPressureDiastolic : BloodPressureValue;
		attribute oxygenSaturation : OxygenSaturationValue;
	}

	// Abstract Sensor Definition
	abstract part def VitalSignSensor {
		port vitalDataOut : VitalSignsData;
	}

	// Concrete Sensor Definitions
	part def HeartRateSensor :> VitalSignSensor {
		attribute currentRate : HeartRateValue;
		flow of vitalDataOut {
			heartRate = currentRate;
		}
	}

	part def BloodPressureSensor :> VitalSignSensor {
		attribute systolic : BloodPressureValue;
		attribute diastolic : BloodPressureValue;
		flow of vitalDataOut {
			bloodPressureSystolic = systolic;
			bloodPressureDiastolic = diastolic;
		}
	}

	part def PulseOximeter :> VitalSignSensor {
		attribute spO2 : OxygenSaturationValue;
		flow of vitalDataOut {
			oxygenSaturation = spO2;
		}
	}

	// Output and Alerting Components
	part def SystemDisplay {
		port displayDataIn : VitalSignsData;
		action renderVitals {
			in data : VitalSignsData;
		}
	}

	part def AlertSystem {
		port alertTrigger;
		state machine AlarmController {
			initial state Idle;
			state Active;

			transition from Idle to Active on receive alertTrigger;
			transition from Active to Idle on after(30[s]);
		}
	}

	// Core Logic Component
	part def ProcessingUnit {
		port hrDataIn : VitalSignsData;
		port bpDataIn : VitalSignsData;
		port spo2DataIn : VitalSignsData;

		port displayDataOut : VitalSignsData;
		port alertCmdOut;

		attribute currentVitals : VitalSignsData;

		// Combine incoming sensor data into a single structure
		action combineSensorData {
			in hr : VitalSignsData from hrDataIn;
			in bp : VitalSignsData from bpDataIn;
			in spo2 : VitalSignsData from spo2DataIn;
			out combined : VitalSignsData to currentVitals;

			then {
				combined.heartRate = hr.heartRate;
				combined.bloodPressureSystolic = bp.bloodPressureSystolic;
				combined.bloodPressureDiastolic = bp.bloodPressureDiastolic;
				combined.oxygenSaturation = spo2.oxygenSaturation;
			}
		}

		// Behavior to monitor vitals and trigger alerts
		state machine MonitorAndAlert {
			initial state NormalMonitoring;
			state Alerting;

			// Define conditions that trigger an alert
			exhibit condition isHeartRateCritical : Boolean =
				currentVitals.heartRate < 50[beatsPerMinute] or
				currentVitals.heartRate > 130[beatsPerMinute];

			exhibit condition isSpo2Critical : Boolean =
				currentVitals.oxygenSaturation < 94[%];

			// Use ChangeSignals to detect when a critical condition becomes true
			exhibit heartRateSignal : ChangeSignal {
				redefines signalCondition = isHeartRateCritical;
			}
			exhibit spo2Signal : ChangeSignal {
				redefines signalCondition = isSpo2Critical;
			}

			// Transition to Alerting state when any critical signal is received
			transition from NormalMonitoring to Alerting
				on receive heartRateSignal or receive spo2Signal;

			// In the Alerting state, send a command to the alert system
			state Alerting {
				entry {
					send alertCmdOut;
				}
				// Return to normal monitoring if conditions stabilize
				transition to NormalMonitoring
					when not (isHeartRateCritical or isSpo2Critical);
			}
		}
	}

	// Top-level System Definition
	part def PatientMonitoringDevice {
		// Component Parts
		part hrSensor : HeartRateSensor;
		part bpSensor : BloodPressureSensor;
		part oximeter : PulseOximeter;
		part processor : ProcessingUnit;
		part display : SystemDisplay;
		part alerter : AlertSystem;

		// Connections between parts
		connect hrSensor.vitalDataOut to processor.hrDataIn;
		connect bpSensor.vitalDataOut to processor.bpDataIn;
		connect oximeter.vitalDataOut to processor.spo2DataIn;
		connect processor.displayDataOut to display.displayDataIn;
		connect processor.alertCmdOut to alerter.alertTrigger;

		// Data flow for the display
		flow from processor.currentVitals to display.displayDataIn;
	}

	// Requirements Definition
	abstract requirement def VitalSignRangeRequirement {
		subject device : PatientMonitoringDevice;
		attribute lowerBound : Real;
		attribute upperBound : Real;
	}

	// Specific System Requirements
	requirement <'PM-R-01'> HeartRateMonitoring : VitalSignRangeRequirement {
		doc "The patient's heart rate shall be maintained and monitored within a safe range."
		subject device : PatientMonitoringDevice;
		redefines lowerBound = 50.0;
		redefines upperBound = 130.0;
		require constraint {
			device.processor.currentVitals.heartRate >= lowerBound[beatsPerMinute] and
			device.processor.currentVitals.heartRate <= upperBound[beatsPerMinute]
		}
	}

	requirement <'PM-R-02'> OxygenSaturationMonitoring {
		doc "The patient's blood oxygen saturation (SpO2) must not fall to critical levels."
		subject device : PatientMonitoringDevice;
		attribute criticalThreshold : Percentage = 94[%];
		require constraint {
			device.processor.currentVitals.oxygenSaturation >= criticalThreshold
		}
	}

	// Verification case to show how requirements are satisfied
	satisfy requirement HeartRateMonitoring by device.processor.MonitorAndAlert;
	satisfy requirement OxygenSaturationMonitoring by device.processor.MonitorAndAlert;
}
