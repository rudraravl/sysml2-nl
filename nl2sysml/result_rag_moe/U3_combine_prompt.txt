Example 1 NL:
This describes a sensor management function within a larger system. This function is responsible for systematically updating a group of sensors.

Each sensor in the group has a status that indicates whether it is "ready" to be updated. A sensor might be considered ready if it is powered on, has completed its self-test, and is not currently busy with another task.

The update process works by checking each sensor in the collection one by one. If a sensor is marked as "ready," the system proceeds to perform an update on it. Example 3 NL:
This model describes an automotive cruise control system designed to automatically maintain a vehicle's speed as set by the driver. The system is composed of several key components working in concert: a central Cruise Controller, a Speedometer, the vehicle's Engine, and the Driver who operates it.

The process is initiated by the Driver, who sends a command to set a desired speed. This command is received by the Cruise Controller, which acts as the brain of the system. To regulate the speed, the controller needs to know how fast the vehicle is currently moving, so it receives a continuous stream of data from the Speedometer.

By comparing the driver's desired speed with the actual speed reported by the speedometer, the Cruise Controller calculates whether to accelerate or decelerate. It then sends a precise fuel command to the Engine, which responds by adjusting its power output to match and maintain the target speed without further input from the driver.

package TemperatureLoggingSystem {
	public import ISQ::*;
	public import Time::*;

	// Value and Item Definitions
	value def Temperature redefines Real {
		unit = ISQ::degreeCelsius;
	}

	item def TemperatureReading {
		attribute value : Temperature;
		attribute timestamp : TimeStamp;
	}

	item def ReadCommand;

	// Interface and Port Definitions
	interface def SensorIO {
		in command : ReadCommand;
		out reading : TemperatureReading;
	}

	interface def LoggerInput {
		in reading : TemperatureReading;
	}

	// Part Definitions
	part def TemperatureSensor {
		port io : SensorIO;
		attribute lastMeasuredTemp : Temperature;

		action measureAndReport {
			out reading : TemperatureReading via io;
			let newReading = new TemperatureReading {
				value = 23.5[°C], // Simulated value
				timestamp = now()
			};
			self.lastMeasuredTemp = newReading.value;
			reading = newReading;
		}

		accept command : ReadCommand via io {
			do measureAndReport;
		}
	}

	part def DataLogger {
		port dataIn : LoggerInput;
		attribute log : TemperatureReading[0..*] ordered;

		action appendToLog {
			in reading : TemperatureReading;
		}

		accept reading : TemperatureReading via dataIn {
			do appendToLog(reading) then {
				self.log = self.log->append(reading);
			}
		}
	}

	part def CentralController {
		part sensor : TemperatureSensor;
		part logger : DataLogger;

		connect sensor.io to logger.dataIn;

		exhibit state logic : ControllerLogic;

		requirement usage loggingFrequency {
			ref def = LoggingFrequencyRequirement;
		}
		satisfy loggingFrequency by self.logic;

		requirement usage dataStorage {
			ref def = DataStorageRequirement;
		}
		satisfy dataStorage by self.logger.log, self.logger.appendToLog;
	}

	// Behavior Definition
	state machine def ControllerLogic {
		inout controller : CentralController;

		initial transition to Idle;

		state Idle;

		transition from Idle to Idle {
			trigger after(5[s]);
			effect send new ReadCommand() to controller.sensor.io;
		}
	}

	// Requirement Definitions
	requirement def LoggingFrequencyRequirement {
		id = "CTL-REQ-01";
		text "The Central Controller shall trigger a temperature reading
		      from the sensor periodically every 5 seconds.";
	}

	requirement def DataStorageRequirement {
		id = "CTL-REQ-02";
		text "All temperature readings produced by the sensor shall be

                        ::temperature = temperature;

                    out temperature : Temperature = 25[degC]; // example temperature
