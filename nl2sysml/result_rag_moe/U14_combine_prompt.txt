		private import DataModel::*;
		private import Behavior::*;

		part def ElevatorControlSystem;

		part def Elevator {
			attribute currentFloor : FloorNumber = 1;
			attribute targetFloor : FloorNumber;
			attribute currentSpeed : Speed = 0 [m/s];

			port requests : FloorRequest[*];

			part controller : ElevatorControlSystem;

			exhibit state machine mainBehavior : ElevatorStateMachine;
		}
	}

	package Behavior {
		private import DataModel::*;
		private import Structure::*;

		state machine def ElevatorStateMachine {
			initial state Idle;
			state Moving;
			state Stopping;
			state DoorsOpen;

			transition from Idle to Moving
				on reception of FloorRequest from Elevator::requests {
				doc "A floor request is received while idle."
				action {
					Elevator::targetFloor = in.destination;
				}
			}

			transition from Moving to Stopping
				when Elevator::currentFloor == Elevator::targetFloor {
				doc "The elevator arrives at the target floor and begins stopping."
			}

			transition from Stopping to DoorsOpen
				when Elevator::currentSpeed == 0 [m/s] {
				doc "The elevator has come to a complete stop."
			}

			transition from DoorsOpen to Idle
				after 5[s] {
				doc "Doors close after a 5-second wait, and the elevator returns to idle."
			}
		}
	}

	package Requirements {
		private import SI::*;
		private import Structure::*;
		private import Behavior::*;

		requirement timelyStop {
			doc "The elevator must come to a complete stop at the requested floor within 3 seconds of initiating the stopping sequence.";

			subject el : Structure::Elevator;

			require constraint {
				duration(el.mainBehavior.Stopping) <= 3[s]
			}
		}
	}

	part mainElevator : Structure::Elevator;

	satisfy Requirements::timelyStop by mainElevator;
